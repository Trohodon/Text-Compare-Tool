Option Compare Database

Private Sub Command18_Click() 'OK Button

    'Declare Variables:
    Dim db As DAO.Database
    Dim SQL As String
    Dim ConstantTypeID As Long
    Dim QDF As DAO.QueryDef
    Dim Constants As DAO.Recordset
    Dim ConstantValue As Double
    Dim Models As Recordset
    Dim ModelID As Long
    Dim MachineID As Long
    Dim ModelType As Long
    Dim Parent As Form

    Dim DateStart As String
    Dim DateEnd As String

    Dim CloneID As Long
    Dim Clone As Boolean

    'TURN OFF WARNINGS:
    DoCmd.SetWarnings False

    'Get Db Handle:
    Set db = CurrentDb

    'Determine next Model ID Number:
    Set Models = db.OpenRecordset("DynamicsModels")
    ModelID = 0
    Models.MoveLast
    Models.MoveFirst
    While Not Models.EOF
        If Models.Fields("model_id").Value > ModelID Then
            ModelID = Models.Fields("model_id").Value
        End If
        Models.MoveNext
    Wend
    ModelID = ModelID + 1

    'Get Data from Form:
    Me.Combo20.SetFocus
    MachineID = Me.Combo20.Value

    Me.Combo22.SetFocus
    ModelType = Me.Combo22.Value

    Me.Check29.SetFocus
    Clone = Me.Check29.Value

    If Clone Then
        Me.Combo27.SetFocus
        CloneID = Me.Combo27.Value
    End If

    Me.Text31.SetFocus
    DateStart = Me.Text31.Text

    Me.Text33.SetFocus
    DateEnd = Me.Text33.Text

    'Check if NULL Form data:
    If MachineID = Null Or ModelType = Null Then

        MsgBox "Cannot add model, Machine ID or Model Type cannot be null"

    Else

        'Insert new Model Record:
        SQL = "INSERT INTO DynamicsModels (model_id, machine_id, model_type, model_date_start, model_date_end) " & _
              "VALUES(" & ModelID & ", " & MachineID & ", " & ModelType & ", #" & DateStart & "#, #" & DateEnd & "#)"

        DoCmd.RunSQL SQL

        'Run Query to get constant field list:
        If Clone Then
            Set QDF = db.QueryDefs("AddModelConstantsClone")
            QDF.Parameters("ModelType") = ModelType
            QDF.Parameters("ModelID") = CloneID
            Set Constants = QDF.OpenRecordset
        Else
            Set QDF = db.QueryDefs("AddModelConstants")
            QDF.Parameters("ModelType") = ModelType
            Set Constants = QDF.OpenRecordset
        End If

        Constants.MoveLast
        Constants.MoveFirst

        'Insert constant fields:
        While Not Constants.EOF

            ConstantTypeID = Constants.Fields("model_type_const_id").Value

            If Clone Then
                ConstantValue = Constants.Fields("value").Value
                SQL = "INSERT INTO DynamicsModelConstants (model_id, model_constant_type_id, const_value) " & _
                      "VALUES(" & ModelID & ", " & ConstantTypeID & ", " & ConstantValue & ")"
            Else
                SQL = "INSERT INTO DynamicsModelConstants (model_id, model_constant_type_id, const_value) " & _
                      "VALUES(" & ModelID & ", " & ConstantTypeID & ", 0)"
            End If

            DoCmd.RunSQL SQL

            Constants.MoveNext
        Wend

        'Close form and report to user:
        DoCmd.Close acForm, "DynamicsAddModelForm"

        If Clone Then
            MsgBox "Model has been added with constant values based on model with ModelID: " & CloneID
        Else
            MsgBox "Model has been added with default constant values. Return to Dynamics Data Edit Form to edit constants."
        End If

        Set Parent = Forms("DynamicsDataEditForm")
        Parent.Requery
        Parent.Refresh

    End If

    'TURN WARNINGS BACK ON:
    DoCmd.SetWarnings True

End Sub


Private Sub Command19_Click() 'Cancel Button

    'Close form without saving:
    DoCmd.Close acForm, "DynamicsAddModelForm", acSaveNo

End Sub
