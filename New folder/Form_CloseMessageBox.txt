'This code is to display a message box when the user closes the database.  I had to create a hidden form to accomplish this.
'There is a blank form called 'CloseMessageBox' in the unassigned objects navigation pane.  If you don't see it grayed out then you
'need to go to file, options, current database, navigation options, and check the box for 'show hidden objects'. This will display the
'hidden form. From here I
Private Sub Form_Unload(Cancel As Integer)
  Dim Answer As Variant

  If Not IsNumeric(Left(Application.CurrentProject.NAME, 4)) Then
  
    Answer = MsgBox("Did you document your changes?", vbYesNo + vbQuestion, "Track Changes")

    'Answer = MDFDMsgBox.MsgBoxSmile ' This doesn't work. I wanted to change the Yes button in the MsgBox to Yes/Didn't Change Anything. Oh well. JWR - 2019/05/09

    If Answer = vbYes Then
        ResetSnapshotsTable

        DoCmd.Quit
    Else
        Cancel = True
        DoCmd.openTable "DocumentChanges", acViewNormal
        Exit Sub
    End If
  End If
End Sub


Private Sub ResetSnapshotsTable()
  Dim snapDate As Date, season As String
  Dim db As DAO.Database
  Dim Snapshots As DAO.Recordset

  DoCmd.SetWarnings False

  Set db = CurrentDb()
  Set Snapshots = db.OpenRecordset("Snapshots", dbOpenDynaset, dbSeeChanges)

  Snapshots.MoveLast
  Snapshots.MoveFirst

  While Not Snapshots.EOF
    If Snapshots.Fields("Selected") Then
      season = Snapshots.Fields("season")
      snapDate = Now

      SQL = "UPDATE Snapshots SET [model_name] = '" & Right(Year(snapDate), 2) & season & "', [year] = '" & Year(snapDate) & "', [date] = '" _
      & month(snapDate) & "/" & day(snapDate) & "/" & Year(snapDate) & "'"

      DoCmd.RunSQL SQL
    Else
      Snapshots.Delete
    End If

    Snapshots.MoveNext
  Wend
  Snapshots.Close
End Sub
